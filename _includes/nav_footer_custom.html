{%- comment -%} Language switcher. English pages live as normal pages under docs/ and keep their original URLs. Chinese pages live in the docs_zh collection and are served from the same path prefixed with /zh. A counterpart therefore differs only by that prefix. Only pages that have actually been translated are linked directly; anything else falls back to the other language's guide index so the link is never broken. {%- endcomment -%} {%- assign is_zh = false -%} {%- if page.url contains '/zh/' -%}{%- assign is_zh = true -%}{%- endif -%} {%- assign target = nil -%} {%- if is_zh -%} {%- assign counterpart = page.url | remove_first: '/zh' -%} {%- for candidate in site.html_pages -%} {%- if candidate.url == counterpart -%}{%- assign target = candidate.url -%}{%- break -%}{%- endif -%} {%- endfor -%} {%- if target == nil -%}{%- assign target = '/guides/' -%}{%- endif -%} {%- else -%} {%- assign counterpart = page.url | prepend: '/zh' -%} {%- for candidate in site.docs_zh -%} {%- if candidate.url == counterpart -%}{%- assign target = candidate.url -%}{%- break -%}{%- endif -%} {%- endfor -%} {%- if target == nil -%}{%- assign target = '/zh/guides/' -%}{%- endif -%} {%- endif -%}
{%- if is_zh %} 中文 English {%- else %} English 中文 {%- endif %}
